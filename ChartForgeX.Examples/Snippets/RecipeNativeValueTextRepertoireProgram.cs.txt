using System;
using System.IO;
using System.Linq;
using System.Text.Json;
using ChartForgeX.Core;
using ChartForgeX.Primitives;
using ChartForgeX.Rendering;
using ChartForgeX.Themes;
using ChartForgeX.Typography;
using ChartForgeX.Interactivity.Html;

// Supply licensed .ttf/.otf files. Both actual faces and native numeric fallback decisions are captured before presentation.
var primaryPath = args.Length > 0 ? args[0] : "fonts/primary.ttf";
var fallbackPath = args.Length > 1 ? args[1] : "fonts/fallback.otf";
foreach (var fontPath in new[] { primaryPath, fallbackPath }) {
    var extension = Path.GetExtension(fontPath); if (!extension.Equals(".ttf", StringComparison.OrdinalIgnoreCase) && !extension.Equals(".otf", StringComparison.OrdinalIgnoreCase))
        throw new ArgumentException("This example accepts explicit .ttf/.otf files.");
}
FontRegistry.Register("Example Primary", primaryPath); FontRegistry.Register("Example Fallback", fallbackPath);
const string family = "Example Primary, Example Fallback";
Chart Model(double value) => NativeRadialMotionRecipe.Create({{DONUT}}, false, value, {{WIDTH}}, VisualThemeMode.{{MODE}})
    .WithFontFamily(family).WithXLabels("Asia", "Europe", "Americas").WithTitle(string.Empty).WithSubtitle(string.Empty).WithHeader(false).WithLegend(false)
    .WithDonutCenterLabel(false).WithPieSliceLabelContent(ChartPieSliceLabelContent.Value).WithDataLabelPlacement(ChartDataLabelPlacement.Inside).WithValueFormat(ChartValueFormat.Compact());
var renderer = new HtmlInteractiveChartRenderer(); var from = renderer.PrepareSection(Model(900)); var to = renderer.PrepareSection(Model(1100));
var options = new HtmlNativeTransitionOptions(HtmlNativeTransitionEffect.Values, 3);
var forward = renderer.PrepareNativeTransitionJson(from, to, options); var reverse = renderer.PrepareNativeTransitionJson(to, from, options);
var action = "const chart=ChartForgeXCharts.get(document.querySelector('.cfx-interactive-chart'));let forward=true;document.querySelector('#play').onclick=()=>{chart.update(forward?"
    + JsonSerializer.Serialize(to.Html) + ":" + JsonSerializer.Serialize(from.Html) + ",{sceneRevision:chart.sceneRevision,transition:forward?"
    + JsonSerializer.Serialize(forward) + ":" + JsonSerializer.Serialize(reverse) + ",motion:{duration:650}});forward=!forward;};";
var css = FontCss("Example Primary", primaryPath) + FontCss("Example Fallback", fallbackPath);
var html = "<!doctype html><html><head><meta charset='utf-8'><meta name='viewport' content='width=device-width,initial-scale=1'><style>" + css + HtmlInteractiveChartRenderer.BuildFragmentStyle()
    + "</style></head><body><button id='play' type='button'>Present captured native captions</button>" + from.Html + "<script>" + HtmlInteractiveChartRenderer.BuildInteractionScript() + "</script><script>" + action + "</script></body></html>";
Directory.CreateDirectory("output"); File.WriteAllText("output/visual.html", html); File.WriteAllText("output/visual.transition.json", forward);
File.WriteAllText("output/visual.svg", to.PreparedVisual.ToSvg()); File.WriteAllBytes("output/visual.png", to.PreparedVisual.ToPng());
static string FontCss(string name, string path) => "@font-face{font-family:'" + name + "';src:url(data:font/" + (Path.GetExtension(path).Equals(".otf", StringComparison.OrdinalIgnoreCase) ? "otf" : "ttf")
    + ";base64," + Convert.ToBase64String(File.ReadAllBytes(path)) + ") format('opentype')}";
