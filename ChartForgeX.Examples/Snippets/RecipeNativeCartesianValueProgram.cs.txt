using System;
using System.IO;
using System.Text.Json;
using ChartForgeX.Core;
using ChartForgeX.Primitives;
using ChartForgeX.Rendering;
using ChartForgeX.Themes;
using ChartForgeX.Interactivity.Html;

var renderer = new HtmlInteractiveChartRenderer();
var from = renderer.PrepareSection(NativeCartesianValueRecipe.Create({{BAR}}, 20, {{WIDTH}}, VisualThemeMode.{{MODE}}));
var to = renderer.PrepareSection(NativeCartesianValueRecipe.Create({{BAR}}, 80, {{WIDTH}}, VisualThemeMode.{{MODE}}));
var initial = {{INITIAL}};
var payload = renderer.PrepareNativeTransitionJson(from, to,
    new HtmlNativeTransitionOptions(initial ? HtmlNativeTransitionEffect.InitialValues : HtmlNativeTransitionEffect.Values, 12));
var action = "const chart=ChartForgeXCharts.get(document.querySelector('.cfx-interactive-chart'));document.querySelector('#play').onclick=()=>{";
action += initial ? "chart.presentInitial(" + JsonSerializer.Serialize(payload) + ",{sceneRevision:chart.sceneRevision,motion:{duration:650}});"
    : "chart.update(" + JsonSerializer.Serialize(to.Html) + ",{sceneRevision:chart.sceneRevision,transition:" + JsonSerializer.Serialize(payload) + ",motion:{duration:650}});document.querySelector('#play').disabled=true;";
action += "};";
var html = "<!doctype html><html><head><meta charset='utf-8'><meta name='viewport' content='width=device-width,initial-scale=1'><style>"
    + HtmlInteractiveChartRenderer.BuildFragmentStyle() + "</style></head><body><button id='play' type='button'>Show native values</button>" + (initial ? to.Html : from.Html)
    + "<script>" + HtmlInteractiveChartRenderer.BuildInteractionScript() + "</script><script>" + action + "</script></body></html>";
Directory.CreateDirectory("output"); File.WriteAllText("output/visual.html", html); File.WriteAllText("output/visual.transition.json", payload);
File.WriteAllText("output/visual.svg", to.PreparedVisual.ToSvg()); File.WriteAllBytes("output/visual.png", to.PreparedVisual.ToPng());
