#region Setup
using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Linq;
using ChartForgeX.Core;
using ChartForgeX.Primitives;
using ChartForgeX.Rendering;
using ChartForgeX.Themes;
using ChartForgeX.Topology;
using ChartForgeX.Typography;
using ChartForgeX.VisualArtifacts;
using ChartForgeX.VisualBlocks;

var mode = VisualThemeMode.Light;
var theme = VisualTheme.Graphite();
#endregion

var capacity = ChartSeries.CreateGauge(new ChartSeriesId("capacity"), "Capacity",
    new ChartGaugeValue(76, targets: new[] { new ChartGaugeTarget(60, "Baseline"), new ChartGaugeTarget(85, "Goal") }));
var chart = Chart.Create().AddSeries(capacity)
    .ConfigureGauge(options => {
        options.Form = ChartGaugeForm.Arc;
options.TickCount = 5;
options.StartAngleDegrees = -90; options.SweepAngleDegrees = 360;                 options.Caption = "Available capacity";
        options.Bands.Add(new ChartGaugeBand(0, 50, ChartSeriesState.Danger));
        options.Bands.Add(new ChartGaugeBand(50, 80, ChartSeriesState.Warning));
        options.Bands.Add(new ChartGaugeBand(80, 100, ChartSeriesState.Success));
    });

#region Output
var context = new VisualRenderContext(
    new VisualLayoutOptions(new VisualSize(800, 440)), theme, mode,
    new VisualFrame("Available capacity against a target", "", showLegend: null, showSurface: false));
var prepared = chart.Prepare(context);
File.WriteAllText("visual.svg", prepared.ToSvg());
File.WriteAllBytes("visual.png", prepared.ToPng());
File.WriteAllText("visual.html", prepared.ToArtifact("example", VisualArtifactKind.Chart).ToHtmlPage());
#endregion
