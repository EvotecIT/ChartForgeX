Create this example

Create a console project, add the ChartForgeX package, and replace Program.cs with this source. Run it to write SVG, PNG and HTML with the default system fonts.

For the same typography as the gallery, download the gallery-font recipe and copy the gallery’s fonts directory beside your program.

#region Setup
using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Linq;
using ChartForgeX.Core;
using ChartForgeX.Primitives;
using ChartForgeX.Rendering;
using ChartForgeX.Themes;
using ChartForgeX.Topology;
using ChartForgeX.Typography;
using ChartForgeX.VisualArtifacts;
using ChartForgeX.VisualBlocks;

var mode = VisualThemeMode.Light;
var theme = VisualTheme.Graphite();
var model = LineRecipe.Create("compact", mode);
var context = new VisualRenderContext(
    new VisualLayoutOptions(new VisualSize(360, 360)), theme, mode,
    new VisualFrame("Weekly request volume", "", showLegend: null, showSurface: false));
var prepared = model.Prepare(context);
File.WriteAllText("visual.svg", prepared.ToSvg());
File.WriteAllBytes("visual.png", prepared.ToPng());
File.WriteAllText("visual.html", prepared.ToArtifact("example", VisualArtifactKind.Chart).ToHtmlPage());
#endregion

/// <summary>Builds request-volume lines and a comparison of step positions and markers.</summary>
public static class LineRecipe {
    /// <summary>Creates the selected line example with its authored observations and styling.</summary>
    public static Chart Create(string variant = "wide", VisualThemeMode mode = VisualThemeMode.Light) {
        if (variant is not ("wide" or "compact" or "sparse" or "options" or "compact-options"))
            throw new ArgumentOutOfRangeException(nameof(variant));
        var chart = Chart.Create();
        if (variant is "options" or "compact-options") {
            chart.WithXLabels("Mon", "Tue", "Wed", "Thu", "Fri").WithYAxis("Recorded level");
            var positions = new[] { ChartStepPosition.Start, ChartStepPosition.Middle, ChartStepPosition.End };
            var shapes = new[] { ChartMarkerShape.Square, ChartMarkerShape.Diamond, ChartMarkerShape.Triangle };
            var ids = new[] { "step-start", "step-middle", "step-end" };
            for (var index = 0; index < positions.Length; index++) {
                var points = new[] { 8d, 17, 11, 23, 16 }.Select((value, point) => new ChartPoint(point + 1, value + index * 30));
                var series = ChartSeries.CreateLine(new ChartSeriesId(ids[index]), positions[index].ToString(), points)
                    .WithInterpolation(ChartInterpolation.Step, positions[index]);
                var shape = shapes[index];
                series.ConfigureMarkers(markers => { markers.Shape = shape; markers.Enabled = true; markers.Radius = 5; });
                chart.AddSeries(series);
            }
        }
        else if (variant == "sparse") {
            var points = Enumerable.Range(1, 6).Where(index => index is 1 or 3 or 6)
                .Select(index => new ChartPoint(index, 20 + index * 7 % 29, index == 6));
            chart.WithXLabels("Mon", "Tue", "Wed", "Thu", "Fri", "Sat")
                .AddSeries(ChartSeries.CreateLine(new ChartSeriesId("requests"), "Requests", points));
        }
        else {
            // Twelve weeks of requests per channel: the portal grows, email declines and the API stays seasonal.
            var weeks = Enumerable.Range(1, 12).ToArray();
            chart.WithXLabels(weeks.Select(week => "W" + week).ToArray()).WithYAxis("Requests");
            chart.AddSeries(ChartSeries.CreateLine(new ChartSeriesId("requests"), "Portal",
                weeks.Select(week => new ChartPoint(week, Math.Round(820 + week * 64 + Math.Sin(week * 1.1) * 60)))));
            chart.AddSeries(ChartSeries.CreateLine(new ChartSeriesId("api"), "API",
                weeks.Select(week => new ChartPoint(week, Math.Round(640 + Math.Sin(week * .7) * 140)))));
            chart.AddSeries(ChartSeries.CreateLine(new ChartSeriesId("email"), "Email",
                weeks.Select(week => new ChartPoint(week, Math.Round(760 - week * 38 + Math.Cos(week * 1.4) * 40)))));
        }
        return chart;
    }
}
