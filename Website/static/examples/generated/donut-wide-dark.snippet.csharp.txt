#region Setup
using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Linq;
using ChartForgeX.Core;
using ChartForgeX.Primitives;
using ChartForgeX.Rendering;
using ChartForgeX.Themes;
using ChartForgeX.Topology;
using ChartForgeX.Typography;
using ChartForgeX.VisualArtifacts;
using ChartForgeX.VisualBlocks;
using static FindingsRecipe;

var mode = VisualThemeMode.Dark;
var theme = VisualTheme.Graphite();
#endregion

var count = Categories.Length;
// Dense findings fall off steeply, as they do in a real assessment: a few categories carry most of the work.
double Count(int index) => Categories[index].Count;
string Name(int index) => Categories[index].Name;
var points = Enumerable.Range(0, count).Where(index => true)
    .Select(index => new ChartPoint(index + 1, Count(index)));
var findings = new ChartSeries(new ChartSeriesId("findings"), "Findings", ChartSeriesKind.Donut, points);
var chart = Chart.Create().WithXLabels(Enumerable.Range(0, count).Select(Name).ToArray()).AddSeries(findings);

#region Output
var context = new VisualRenderContext(
    new VisualLayoutOptions(new VisualSize(800, 440)), theme, mode,
    new VisualFrame("Findings by category", "Open findings from the October assessment", showLegend: null, showSurface: false));
var prepared = chart.Prepare(context);
File.WriteAllText("visual.svg", prepared.ToSvg());
File.WriteAllBytes("visual.png", prepared.ToPng());
File.WriteAllText("visual.html", prepared.ToArtifact("example", VisualArtifactKind.Chart).ToHtmlPage());
#endregion

static class FindingsRecipe {
    internal static readonly (string Name, string Long, double Count)[] Categories = {
        ("Patching", "Missing security updates on servers", 46), ("Identity", "Stale and over-privileged accounts", 38),
        ("Certificates", "Certificates expiring within 30 days", 27), ("Network", "Exposed management ports", 21),
        ("Backups", "Backups without a recent restore test", 17), ("Logging", "Audit logging disabled or incomplete", 12),
        ("Encryption", "Unencrypted data at rest", 8)
    };
}
