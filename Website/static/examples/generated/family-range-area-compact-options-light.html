Create a console project, add the ChartForgeX package, and replace Program.cs with this source. Run it to write SVG, PNG and HTML with the default system fonts.

For the same typography as the gallery, download the gallery-font recipe and copy the gallery’s fonts directory beside your program.

#region Setup
using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Linq;
using ChartForgeX.Core;
using ChartForgeX.Primitives;
using ChartForgeX.Rendering;
using ChartForgeX.Themes;
using ChartForgeX.Topology;
using ChartForgeX.Typography;
using ChartForgeX.VisualArtifacts;
using ChartForgeX.VisualBlocks;

var mode = VisualThemeMode.Light;
var theme = VisualTheme.Graphite();
var model = RangeAreaRecipe.Create("compact-options", mode);
var context = new VisualRenderContext(
    new VisualLayoutOptions(new VisualSize(360, 360)), theme, mode,
    new VisualFrame("Expected operating range", "Lower, middle and upper bounds share middle-step transitions", showLegend: null, showSurface: false));
var prepared = model.Prepare(context);
File.WriteAllText("visual.svg", prepared.ToSvg());
File.WriteAllBytes("visual.png", prepared.ToPng());
File.WriteAllText("visual.html", prepared.ToArtifact("example", VisualArtifactKind.Chart).ToHtmlPage());
#endregion

/// <summary>Builds an expected operating range with smooth or stepped area boundaries.</summary>
public static class RangeAreaRecipe {
    /// <summary>Creates the selected range-area example from typed lower and upper limits.</summary>
    public static Chart Create(string variant = "wide", VisualThemeMode mode = VisualThemeMode.Light) {
        if (variant is not ("wide" or "compact" or "sparse" or "options" or "compact-options" or "singleton" or "compact-singleton" or "independent-boundaries" or "compact-independent-boundaries"))
            throw new ArgumentOutOfRangeException(nameof(variant));
        var chart = Chart.Create();
        if (variant is "independent-boundaries" or "compact-independent-boundaries") {
            var ranges = ChartRangeBands.FromXY(new[] { 1d, 2, 3, 4, 5, 6, 7, 8 },
                new double?[] { 12, 22, 16, null, 25, 15, 20, 18 }, new double?[] { 32, 46, 38, 40, 48, 41, 52, 44 });
            var expected = ChartSeries.CreateRangeArea(new ChartSeriesId("expected-range"), "Expected range", ranges)
                .WithBoundaryInterpolation(new(ChartInterpolation.Smooth), new(ChartInterpolation.Step, ChartStepPosition.Middle))
                .WithObservationId(3, new ChartObservationId("resumed-range"));
            chart.WithTitle("Smooth lower and held upper limits").WithXAxis("Observation").WithYAxis("Expected level").AddSeries(expected);
        }
        else if (variant is "singleton" or "compact-singleton") {
            chart.WithTitle("One observed range").AddSeries(ChartSeries.CreateRangeArea(new ChartSeriesId("expected-range"), "Expected range",
                new[] { new ChartRangeBand(0, 0, 7) }).WithObservationId(0, new ChartObservationId("only-range")));
        }
        else if (variant is "options" or "compact-options") {
            var expected = ChartSeries.CreateRangeArea(new ChartSeriesId("expected-range"), "Expected range", new[] {
                new ChartRangeBand(1, 15, 30), new ChartRangeBand(2, 25, 50), new ChartRangeBand(3, 20, 40),
                new ChartRangeBand(4, 40, 65), new ChartRangeBand(5, 30, 55)
            }).WithInterpolation(ChartInterpolation.Step, ChartStepPosition.Middle);
            chart.WithXLabels("Mon", "Tue", "Wed", "Thu", "Fri").WithYAxis("Expected level").AddSeries(expected);
        }
        else {
            var count = variant == "sparse" ? 2 : 5;
            var ranges = Enumerable.Range(1, count).Select(index => new ChartRangeBand(index, 20 + index * 2, 40 + index * 4));
            var expected = ChartSeries.CreateRangeArea(new ChartSeriesId("expected-range"), "Expected range", ranges);
            chart.WithXLabels("08:00", "10:00", "12:00", "14:00", "16:00").AddSeries(expected);
        }
        return chart;
    }
}
