#region Setup
using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Linq;
using ChartForgeX.Core;
using ChartForgeX.Primitives;
using ChartForgeX.Rendering;
using ChartForgeX.Themes;
using ChartForgeX.Topology;
using ChartForgeX.Typography;
using ChartForgeX.VisualArtifacts;
using ChartForgeX.VisualBlocks;
using static RadialColumnRecipe;

var mode = VisualThemeMode.Dark;
var theme = VisualTheme.Graphite();
#endregion

var chart = AllLabels();

#region Output
var context = new VisualRenderContext(
    new VisualLayoutOptions(new VisualSize(800, 440)), theme, mode,
    new VisualFrame("Regional service workload", "All captions fit individually; caption overlaps are intentional", showLegend: null, showSurface: false));
var prepared = chart.Prepare(context);
File.WriteAllText("visual.svg", prepared.ToSvg());
File.WriteAllBytes("visual.png", prepared.ToPng());
File.WriteAllText("visual.html", prepared.ToArtifact("example", VisualArtifactKind.Chart).ToHtmlPage());
#endregion

static class RadialColumnRecipe {
    internal static Chart AllLabels() {
        var requests = new ChartSeries(new ChartSeriesId("requests"), "Requests", ChartSeriesKind.RadialColumn,
            Enumerable.Range(1, 20).Select(index => new ChartPoint(index, 500 + index * 20)));
        var chart = Chart.Create().WithXLabels(Enumerable.Range(1, 20).Select(index => "Region " + index + " request volume").ToArray())
            .WithRadialGeometry(new ChartRadialGeometryOptions(-75, 195, .2, .25, .1))
            .WithYAxisBounds(0, 1500).WithDataLabels(false).WithXAxis("All region captions; overlaps are intentional")
            .ConfigureXAxis(axis => axis.LabelDensity = ChartLabelDensity.All).AddSeries(requests);
        chart.Options.YAxis.Visible = false;
        return chart;
    }
}
