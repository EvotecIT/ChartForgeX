#region Setup
using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Linq;
using ChartForgeX;
using ChartForgeX.Core;
using ChartForgeX.Primitives;
using ChartForgeX.Rendering;
using ChartForgeX.Themes;
using ChartForgeX.Topology;
using ChartForgeX.Typography;
using ChartForgeX.VisualArtifacts;
using ChartForgeX.VisualBlocks;

var mode = ThemeMode.Dark;
var theme = VisualTheme.Graphite();
#endregion

var chart = Chart.Create();
var movement = ChartSeries.CreateWaterfall(new ChartSeriesId("movement"), "Movement", new[] {
    ChartWaterfallItem.OpeningBalance(1, 120), ChartWaterfallItem.Delta(2, -30), ChartWaterfallItem.Subtotal(3),
    ChartWaterfallItem.Delta(4, -130), ChartWaterfallItem.Subtotal(5), ChartWaterfallItem.Delta(6, 70),
    ChartWaterfallItem.Total(7)
}).ObservationId(3, new ChartObservationId("cost-change"))
    .ObservationId(4, new ChartObservationId("cost-checkpoint"))
    .PointColor(4, ChartColor.FromHex("#A5358A"));
chart.XLabels("Open", "Adjust", "Stage", "Costs", "Cost sum", "Recover", "Balance").AddSeries(movement)
    .Waterfall(options => options.Orientation = ChartOrientation.Horizontal);
chart.XAxis(x => x.Title("Running balance")).YAxis(y => y.Title("Stage")).Bars(b => b.Style(ChartBarStyle.SegmentedCapsule));
chart.Options.XAxis.Reversed(false);
chart.Options.YAxis.Reversed();
chart.DataLabels();

#region Output
var context = new VisualRenderContext(
    new VisualLayoutOptions(new VisualSize(800, 440)), theme, mode,
    new VisualFrame("Changes in available capacity", "Horizontal changes and checkpoints share a cumulative value axis", showLegend: null, showSurface: false));
var prepared = chart.Prepare(context);
File.WriteAllText("visual.svg", prepared.ToSvg());
File.WriteAllBytes("visual.png", prepared.ToPng());
File.WriteAllText("visual.html", prepared.ToArtifact("example", VisualArtifactKind.Chart).ToHtmlPage());
#endregion
