#region Setup
using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Linq;
using ChartForgeX;
using ChartForgeX.Core;
using ChartForgeX.Primitives;
using ChartForgeX.Rendering;
using ChartForgeX.Themes;
using ChartForgeX.Topology;
using ChartForgeX.Typography;
using ChartForgeX.VisualArtifacts;
using ChartForgeX.VisualBlocks;
using static BoxPlotRecipe;

var mode = ThemeMode.Dark;
var theme = VisualTheme.Graphite();
#endregion

var chart = Chart.Create();
chart = StatisticLabels(mode);

#region Output
var context = new VisualRenderContext(
    new VisualLayoutOptions(new VisualSize(360, 360)), theme, mode,
    new VisualFrame("Delivery time by service level", "", showLegend: null, showSurface: false));
var prepared = chart.Prepare(context);
File.WriteAllText("visual.svg", prepared.ToSvg());
File.WriteAllBytes("visual.png", prepared.ToPng());
File.WriteAllText("visual.html", prepared.ToArtifact("example", VisualArtifactKind.Chart).ToHtmlPage());
#endregion

static class BoxPlotRecipe {
    internal static Chart StatisticLabels(ThemeMode mode) {
        var chart = Chart.Create().Title("Five-number delivery summaries").XLabels("Standard", "Express")
            .AddSeries(ChartSeries.CreateBoxPlot(new ChartSeriesId("delivery"), "Delivery time", new[] {
                new ChartBoxPlot(1, 10, 20, 30, 40, 50), new ChartBoxPlot(2, 5, 15, 25, 35, 45)
            }).PointLabel(0, "Standard delivery summary"))
            .BoxPlot(options => {
                options.Orientation = ChartOrientation.Horizontal;
                options.LabelContent = ChartBoxPlotLabelContent.Statistics;
                foreach (var label in new[] { options.MinimumLabel, options.Q1Label, options.MedianLabel, options.Q3Label, options.MaximumLabel }) {
                    label.Placement = ChartDataLabelPlacement.Above;
                    label.Style.FontSize(10);
                }
                options.MinimumLabel.Style.Color(ChartColor.FromHex(mode == ThemeMode.Dark ? "#82b8ff" : "#235b9f"));
                options.MaximumLabel.Style.Color(ChartColor.FromHex(mode == ThemeMode.Dark ? "#ffbe79" : "#975000"));
                options.MedianLabel.Style.FontSize(12).FontWeight(700);
            }).DataLabels();
        chart.Options.XAxis.Bounds(0, 60);
        return chart.XAxis(x => x.Title("Delivery time (minutes)")).YAxis(y => y.Title("Service"));
    }
}
