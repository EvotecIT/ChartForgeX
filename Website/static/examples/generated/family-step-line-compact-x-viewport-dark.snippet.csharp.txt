#region Setup
using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Linq;
using ChartForgeX;
using ChartForgeX.Core;
using ChartForgeX.Primitives;
using ChartForgeX.Rendering;
using ChartForgeX.Themes;
using ChartForgeX.Topology;
using ChartForgeX.Typography;
using ChartForgeX.VisualArtifacts;
using ChartForgeX.VisualBlocks;
using static StepLineViewportRecipe;

var mode = ThemeMode.Dark;
var theme = VisualTheme.Graphite();
#endregion

var observed = new ChartSeries(new ChartSeriesId("observed"), "Observed", ChartSeriesKind.StepLine,
    Points(88, 44, 34, 56, 46, 64, 42, 58, 50, 38, 82)).Markers(markers => markers.Radius = 3);
observed.Interpolation(ChartInterpolation.Step, ChartStepPosition.Middle);
var chart = Chart.Create().XAxis(x => x.Title("Sample")).YAxis(y => y.Title("Requests")).XAxis(x => x.Bounds(0, 10)).DataLabels().AddSeries(observed);
chart.Options.XAxis.Reversed();
chart.Options.XAxis.TickCount = 4;
chart.Options.YAxis.TickCount = 4;

#region Output
var context = new VisualRenderContext(
    new VisualLayoutOptions(new VisualSize(360, 360)), theme, mode,
    new VisualFrame("Reversed steps, X 2\u20138", "Full-data Y; gap at 6; labels in X 2\u20138", showLegend: false, showSurface: false));
var prepared = chart.Prepare(context, new ChartCartesianViewport(x: new ChartAxisWindow(2, 8)));
File.WriteAllText("visual.svg", prepared.ToSvg());
File.WriteAllBytes("visual.png", prepared.ToPng());
File.WriteAllText("visual.html", prepared.ToArtifact("example", VisualArtifactKind.Chart).ToHtmlPage());
#endregion

static class StepLineViewportRecipe {
    internal static ChartPoint[] Points(params double[] values) => values.Select((value, index) =>
        new ChartPoint(index == 1 ? 1.99 : index == 9 ? 8.01 : index, value, index == 6)).ToArray();
}
