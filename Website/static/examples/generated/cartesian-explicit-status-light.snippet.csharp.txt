#region Setup
using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Linq;
using ChartForgeX.Core;
using ChartForgeX.Primitives;
using ChartForgeX.Rendering;
using ChartForgeX.Themes;
using ChartForgeX.Topology;
using ChartForgeX.Typography;
using ChartForgeX.VisualArtifacts;
using ChartForgeX.VisualBlocks;
using static ComparisonRecipe;

var mode = VisualThemeMode.Light;
var theme = VisualTheme.Graphite();
#endregion

var count = 12;
var names = new[] { "Observed", "Expected", "Capacity" };
// Bars sit behind the forecast area and the observed line, so the series are added back to front.
var capacity = ChartSeries.CreateBar(new ChartSeriesId("capacity"), names[2], Points(count, 2, false));
var expected = ChartSeries.CreateArea(new ChartSeriesId("expected"), names[1], Points(count, 1));
var observed = ChartSeries.CreateLine(new ChartSeriesId("observed"), names[0], Points(count, 0));
var colors = VisualTheme.Graphite().Resolve(mode);
var selected = new[] { colors.Status.Pass.Fill, colors.Status.Medium.Fill, colors.Status.Critical.Fill };
observed.WithColor(selected[0]);
expected.WithColor(selected[1]);
capacity.WithColor(selected[2]);
var chart = Chart.Create().WithXLabels(Enumerable.Range(1, count).Select(index => "Day " + index).ToArray())
    .AddSeries(capacity).AddSeries(expected).AddSeries(observed);

#region Output
var context = new VisualRenderContext(
    new VisualLayoutOptions(new VisualSize(800, 440)), theme, mode,
    new VisualFrame("Checks over the reporting period", "Observed checks against the forecast and available capacity", showLegend: null, showSurface: false));
var prepared = chart.Prepare(context);
File.WriteAllText("visual.svg", prepared.ToSvg());
File.WriteAllBytes("visual.png", prepared.ToPng());
File.WriteAllText("visual.html", prepared.ToArtifact("example", VisualArtifactKind.Chart).ToHtmlPage());
#endregion

static class ComparisonRecipe {
    internal static IEnumerable<ChartPoint> Points(int count, int series, bool allowBreaks = true) {
        for (var index = 0; index < count; index++) {
            // Capacity grows in steps, the forecast follows it and observed checks swing around the forecast.
            var value = series switch {
                2 => 64 + index / 3 * 5 + index % 3,
                1 => 46 + index * 1.6,
                _ => 44 + index * 1.7 + Math.Round(Math.Sin(index * 1.3) * 7)
            };
            yield return new ChartPoint(index + 1, Math.Round(value), false);
        }
    }
}
