using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Linq;
using ChartForgeX.Core;
using ChartForgeX.Primitives;
using ChartForgeX.Rendering;
using ChartForgeX.Themes;
using ChartForgeX.Topology;
using ChartForgeX.Typography;
using ChartForgeX.VisualArtifacts;
using ChartForgeX.VisualBlocks;

// Optional exact gallery typography; copy the gallery's fonts directory beside this program.
FontRegistry.Register("CFX Proof Carlito", "fonts/Carlito-Regular.ttf", 400);
FontRegistry.Register("CFX Proof Carlito", "fonts/Carlito-Bold.ttf", 700);
var mode = VisualThemeMode.Light;
var theme = VisualTheme.Graphite();
var model = StateTimelineRecipe.Create("compact", mode);
if ((object)model is TopologyChart topology && topology.DefaultRenderOptions?.FitContentToViewport == false) {
    var natural = topology.WithTitle("Availability throughout the day").WithSubtitle("The same data in a compact view").WithTheme(theme.WithFontFamily("CFX Proof Carlito"), mode).ToVisualArtifact();
    File.WriteAllText("visual.svg", natural.ToSvg());
    File.WriteAllBytes("visual.png", natural.ToPng());
    File.WriteAllText("visual.html", natural.ToHtmlPage(new VisualArtifactRenderOptions { HtmlSizing = VisualArtifactHtmlSizing.PreserveSize }));
    return;
}
var context = new VisualRenderContext(
    new VisualLayoutOptions(new VisualSize(360, 360)), theme, mode,
    new VisualFrame("Availability throughout the day", "The same data in a compact view", showLegend: null, showSurface: false), FontSpec.FromFamily("CFX Proof Carlito"));
var prepared = model.Prepare(context);
File.WriteAllText("visual.svg", prepared.ToSvg());
File.WriteAllBytes("visual.png", prepared.ToPng());
File.WriteAllText("visual.html", prepared.ToArtifact("example", VisualArtifactKind.Chart).ToHtmlPage());

/// <summary>Availability intervals retain state keys, details and lane grouping.</summary>
public static class StateTimelineRecipe {
    public static Chart Create(string variant = "wide", VisualThemeMode mode = VisualThemeMode.Light) {
        var colors = VisualTheme.Graphite().Resolve(mode);
        var start = new DateTime(2026, 3, 2, 8, 0, 0, DateTimeKind.Utc);
        var primary = ChartSeries.CreateStateTimelineLane(new ChartSeriesId("primary"), "Primary", new[] {
            new ChartStateTimelineSegment(start, start.AddHours(3), "pass"),
            new ChartStateTimelineSegment(start.AddHours(3), start.AddHours(5), "warning", "Increased response time"),
            new ChartStateTimelineSegment(start.AddHours(6), start.AddHours(10), "pass")
        }, group: "Production");
        var secondary = ChartSeries.CreateStateTimelineLane(new ChartSeriesId("secondary"), "Secondary", new[] {
            new ChartStateTimelineSegment(start, start.AddHours(6), "pass"),
            new ChartStateTimelineSegment(start.AddHours(7), start.AddHours(10), "critical", "No response")
        }, group: "Production");
        return Chart.Create().WithStateCategories(new ChartStateCategory("pass", "Available", colors.Status.Pass.Fill),
            new ChartStateCategory("warning", "Review", colors.Status.Medium.Fill, ChartStatePattern.Hatched),
            new ChartStateCategory("critical", "Unavailable", colors.Status.Critical.Fill, ChartStatePattern.CrossHatched))
            .AddSeries(primary).AddSeries(secondary).WithValueFormat(ChartValueFormat.Number("0.##", CultureInfo.InvariantCulture));
    }
}
