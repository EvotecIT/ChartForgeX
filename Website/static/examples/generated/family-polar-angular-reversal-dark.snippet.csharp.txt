#region Setup
using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Linq;
using ChartForgeX.Core;
using ChartForgeX.Primitives;
using ChartForgeX.Rendering;
using ChartForgeX.Themes;
using ChartForgeX.Topology;
using ChartForgeX.Typography;
using ChartForgeX.VisualArtifacts;
using ChartForgeX.VisualBlocks;
using static PolarRecipe;

var mode = VisualThemeMode.Dark;
var theme = VisualTheme.Graphite();
#endregion

var chart = Reversed();

#region Output
var context = new VisualRenderContext(
    new VisualLayoutOptions(new VisualSize(800, 440)), theme, mode,
    new VisualFrame("Directional sensor readings", "Clockwise authored radians retain their source directions and values", showLegend: null, showSurface: false));
var prepared = chart.Prepare(context);
File.WriteAllText("visual.svg", prepared.ToSvg());
File.WriteAllBytes("visual.png", prepared.ToPng());
File.WriteAllText("visual.html", prepared.ToArtifact("example", VisualArtifactKind.Chart).ToHtmlPage());
#endregion

static class PolarRecipe {
    internal static Chart Reversed() {
        var angles = new[] { 0d, .55, 1.35, 2.8, 3.65, 5.4 };
        var readings = new[] { 84d, 62, 96, 58, 78, 69 };
        var expected = new[] { 72d, 76, 82, 70, 74, 80 };
        var observed = new ChartSeries(new ChartSeriesId("observed"), "Observed", ChartSeriesKind.Polar,
            angles.Select((angle, index) => new ChartPoint(angle, readings[index])));
        var target = new ChartSeries(new ChartSeriesId("target"), "Expected", ChartSeriesKind.Polar,
            angles.Select((angle, index) => new ChartPoint(angle, expected[index])));
        return Chart.Create().WithYAxisBounds(0, 100).WithDataLabels(false)
            .ConfigureXAxis(axis => { axis.LabelFormatter = Degrees; axis.WithReversal(); })
            .AddSeries(observed).AddSeries(target);
    }

    internal static string Degrees(double angle) => (angle * 180 / Math.PI).ToString("0", System.Globalization.CultureInfo.InvariantCulture) + "°";
}
