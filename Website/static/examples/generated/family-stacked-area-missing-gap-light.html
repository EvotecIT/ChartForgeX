#region Setup
using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Linq;
using ChartForgeX.Core;
using ChartForgeX.Primitives;
using ChartForgeX.Rendering;
using ChartForgeX.Themes;
using ChartForgeX.Topology;
using ChartForgeX.Typography;
using ChartForgeX.VisualArtifacts;
using ChartForgeX.VisualBlocks;

var mode = VisualThemeMode.Light;
var theme = VisualTheme.Graphite();
var model = StackedAreaRecipe.Create("missing-gap", mode);
var context = new VisualRenderContext(
    new VisualLayoutOptions(new VisualSize(800, 440)), theme, mode,
    new VisualFrame("Requests and follow-ups over time", "", showLegend: null, showSurface: false));
var prepared = model.Prepare(context);
File.WriteAllText("visual.svg", prepared.ToSvg());
File.WriteAllBytes("visual.png", prepared.ToPng());
File.WriteAllText("visual.html", prepared.ToArtifact("example", VisualArtifactKind.Chart).ToHtmlPage());
#endregion

/// <summary>Builds stacked request volumes and normalized stepped capacity shares.</summary>
public static class StackedAreaRecipe {
    /// <summary>Creates the selected stacked-area example.</summary>
    public static Chart Create(string variant = "wide", VisualThemeMode mode = VisualThemeMode.Light) {
        if (variant is not ("wide" or "compact" or "sparse" or "options" or "compact-options" or "missing-zero" or "compact-missing-zero" or "missing-gap" or "compact-missing-gap"))
            throw new ArgumentOutOfRangeException(nameof(variant));
        var chart = Chart.Create();
        if (variant is "missing-zero" or "compact-missing-zero" or "missing-gap" or "compact-missing-gap") {
            var policy = variant.EndsWith("gap", StringComparison.Ordinal) ? ChartStackedAreaMissingDataPolicy.Gap : ChartStackedAreaMissingDataPolicy.Zero;
            var received = new ChartSeries(new ChartSeriesId("received"), "Received", ChartSeriesKind.StackedArea,
                new[] { new ChartPoint(1, 18), new ChartPoint(2, 25), new ChartPoint(4, 0), new ChartPoint(5, 20), new ChartPoint(6, 28) })
                .WithStackedAreaMissingData(policy).WithObservationId(2, new ChartObservationId("observed-zero"));
            var processed = new ChartSeries(new ChartSeriesId("processed"), "Processed", ChartSeriesKind.StackedArea,
                new[] { new ChartPoint(1, 12), new ChartPoint(2, 14), new ChartPoint(3, 16), new ChartPoint(4, 9), new ChartPoint(5, 11), new ChartPoint(6, 15) })
                .WithStackedAreaMissingData(policy);
            chart.WithTitle(policy == ChartStackedAreaMissingDataPolicy.Gap ? "Missing contribution cuts the stack" : "Missing contribution has zero height")
                .WithXLabels("Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday").WithYAxis("Requests")
                .AddSeries(received).AddSeries(processed);
        }
        else if (variant is "options" or "compact-options") {
            var completed = new ChartSeries(new ChartSeriesId("completed"), "Completed", ChartSeriesKind.StackedArea,
                new[] { 45d, 70, 80, 55, 90 }.Select((value, point) => new ChartPoint(point + 1, value)))
                .WithNormalization(100).WithInterpolation(ChartInterpolation.Step, ChartStepPosition.Middle);
            var remaining = new ChartSeries(new ChartSeriesId("remaining"), "Remaining", ChartSeriesKind.StackedArea,
                new[] { 55d, 45, 20, 45, 30 }.Select((value, point) => new ChartPoint(point + 1, value)))
                .WithNormalization(100).WithInterpolation(ChartInterpolation.Step, ChartStepPosition.Middle);
            chart.WithXLabels("Mon", "Tue", "Wed", "Thu", "Fri").WithYAxis("Share (%)")
                .AddSeries(completed).AddSeries(remaining);
            chart.Options.YAxis.WithBounds(0, 100).WithLabelFormatter(Percent);
        }
        else {
            var requests = new ChartSeries(new ChartSeriesId("requests"), "Requests", ChartSeriesKind.StackedArea, Observations(variant, 0));
            var followUps = new ChartSeries(new ChartSeriesId("follow-ups"), "Follow-ups", ChartSeriesKind.StackedArea, Observations(variant, -10));
            chart.WithXLabels("Mon", "Tue", "Wed", "Thu", "Fri", "Sat").AddSeries(requests).AddSeries(followUps);
        }
        return chart;
    }

    private static ChartPoint[] Observations(string variant, double offset) =>
        Enumerable.Range(1, 6).Where(index => variant != "sparse" || index is 1 or 3 or 6)
            .Select(index => new ChartPoint(index, 20 + offset + index * 7 % 29, variant == "sparse" && index == 6)).ToArray();

    private static string Percent(double value) => value.ToString("0", CultureInfo.InvariantCulture) + "%";
}
