#region Setup
using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Linq;
using ChartForgeX.Core;
using ChartForgeX.Primitives;
using ChartForgeX.Rendering;
using ChartForgeX.Themes;
using ChartForgeX.Topology;
using ChartForgeX.Typography;
using ChartForgeX.VisualArtifacts;
using ChartForgeX.VisualBlocks;

var mode = VisualThemeMode.Dark;
var theme = VisualTheme.Graphite();
#endregion

var day = new DateTime(2026, 3, 1, 0, 0, 0, DateTimeKind.Utc);
var actual = new[] { 42d, 68, 78, 60, 74, 46, 62 };
var forecast = new[] { 56d, 54, 62, 60, 58, 64, 50 };
var pairs = new ChartDifferencePoint[actual.Length];
for (var item = 0; item < pairs.Length; item++) pairs[item] = new ChartDifferencePoint(item + 1, actual[item], forecast[item], item == 5);
var chart = Chart.Create().WithXAxis("Observation")
    .WithYAxis("Completed requests");
var comparison = ChartSeries.CreateDifferenceArea(new ChartSeriesId("comparison"), "Actual and forecast", pairs);
comparison.ConfigureDifferenceArea(options => {
    options.FirstName = "Actual"; options.SecondName = "Forecast";
    options.PositiveFill = ChartColor.FromHex("#267EAB"); options.NegativeFill = ChartColor.FromHex("#D47A34");
    options.PositiveFillOpacity = .48; options.NegativeFillOpacity = .48;
}).ConfigureMarkers(markers => markers.Radius = 3);
comparison.WithInterpolation(ChartInterpolation.Step, ChartStepPosition.End);
chart.AddSeries(comparison);

#region Output
var context = new VisualRenderContext(
    new VisualLayoutOptions(new VisualSize(800, 440)), theme, mode,
    new VisualFrame("Actual and forecast requests", "Paired values change at the following observation; missing sections stay open", showLegend: null, showSurface: false));
var prepared = chart.Prepare(context);
File.WriteAllText("visual.svg", prepared.ToSvg());
File.WriteAllBytes("visual.png", prepared.ToPng());
File.WriteAllText("visual.html", prepared.ToArtifact("example", VisualArtifactKind.Chart).ToHtmlPage());
#endregion
