using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Linq;
using ChartForgeX.Core;
using ChartForgeX.Primitives;

var mode = VisualThemeMode.Light;
var theme = VisualTheme.Graphite();
var model = RangeBandRecipe.Create("dense", mode);
if ((object)model is TopologyChart topology && topology.DefaultRenderOptions?.FitContentToViewport == false) {
    var natural = topology.WithTitle("Expected temperature band").WithSubtitle("Explore the data, then download the chart").WithTheme(theme, mode).ToVisualArtifact();
    File.WriteAllText("visual.svg", natural.ToSvg());
    File.WriteAllBytes("visual.png", natural.ToPng());
    File.WriteAllText("visual.html", natural.ToHtmlPage(new VisualArtifactRenderOptions { HtmlSizing = VisualArtifactHtmlSizing.PreserveSize }));
    return;
}
var context = new VisualRenderContext(
    new VisualLayoutOptions(new VisualSize(760, 440)), theme, mode,
    new VisualFrame("Expected temperature band", "Explore the data, then download the chart", showLegend: null, showSurface: false));
var prepared = model.Prepare(context);
File.WriteAllText("visual.svg", prepared.ToSvg());
File.WriteAllBytes("visual.png", prepared.ToPng());
File.WriteAllText("visual.html", prepared.ToArtifact("example", VisualArtifactKind.Chart).ToHtmlPage());

/// <summary>Builds an expected temperature band from typed lower and upper limits.</summary>
public static class RangeBandRecipe {
    /// <summary>Creates the selected range-band example.</summary>
    public static Chart Create(string variant = "wide", VisualThemeMode mode = VisualThemeMode.Light) {
        if (variant is not ("wide" or "compact" or "sparse" or "options" or "compact-options" or "singleton" or "compact-singleton" or "dense" or "compact-dense"))
            throw new ArgumentOutOfRangeException(nameof(variant));
        if (variant is "singleton" or "compact-singleton") return Chart.Create().WithTitle("One observed range")
            .AddSeries(ChartSeries.CreateRangeBand(new ChartSeriesId("expected-range"), "Expected range", new[] { new ChartRangeBand(0, 0, 7) })
                .WithObservationId(0, new ChartObservationId("only-range")));
        if (variant is "dense" or "compact-dense") return Chart.Create().WithTitle("Estimated uncertainty").WithXAxis("Observation").WithYAxis("Expected level")
            .AddSeries(ChartSeries.CreateRangeBand(new ChartSeriesId("expected-range"), "Expected range",
                Enumerable.Range(0, 32).Select(index => new ChartRangeBand(index,
                    index % 7 == 0 ? 0 : Math.Sin(index * .5) * 9,
                    index % 7 == 0 ? 0 : Math.Sin(index * .5) * 9 + 4, index == 16))));
        var count = variant == "sparse" ? 2 : 5;
        var ranges = Enumerable.Range(1, count).Select(index => new ChartRangeBand(index, 20 + index * 2, 40 + index * 4));
        var expected = ChartSeries.CreateRangeBand(new ChartSeriesId("expected-range"), "Expected range", ranges);
        return Chart.Create().WithXLabels("08:00", "10:00", "12:00", "14:00", "16:00").AddSeries(expected)
            .WithValueFormat(ChartValueFormat.Number("0.##", CultureInfo.InvariantCulture));
    }
}
