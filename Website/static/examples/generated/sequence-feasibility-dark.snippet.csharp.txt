#region Setup
using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Linq;
using ChartForgeX.Core;
using ChartForgeX.Primitives;
using ChartForgeX.Rendering;
using ChartForgeX.Themes;
using ChartForgeX.Topology;
using ChartForgeX.Typography;
using ChartForgeX.VisualArtifacts;
using ChartForgeX.VisualBlocks;

var mode = VisualThemeMode.Dark;
var theme = VisualTheme.Graphite();
#endregion

var chart = SequenceArtifact.Create("session").WithSize(600, 375)
    .AddParticipant("client", "Client")
    .AddParticipant("server", "Server")
    .AddMessage("client", "server", "Request\npayload")
    .AddMessage("server", "client", "Response", SequenceArtifactMessageLineStyle.Dashed, SequenceArtifactMessageKind.Return);

#region Output
var context = new VisualRenderContext(
    new VisualLayoutOptions(new VisualSize(600, 375)), theme, mode,
    new VisualFrame("Client and server exchange", "Product-neutral diagram semantics", showLegend: false, showSurface: false));
var prepared = chart.Prepare(context);
File.WriteAllText("visual.svg", prepared.ToSvg());
File.WriteAllBytes("visual.png", prepared.ToPng());
File.WriteAllText("visual.html", prepared.ToArtifact("example", VisualArtifactKind.Sequence).ToHtmlPage());
#endregion
