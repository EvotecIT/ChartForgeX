#region Setup
using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Linq;
using ChartForgeX;
using ChartForgeX.Core;
using ChartForgeX.Primitives;
using ChartForgeX.Rendering;
using ChartForgeX.Themes;
using ChartForgeX.Topology;
using ChartForgeX.Typography;
using ChartForgeX.VisualArtifacts;
using ChartForgeX.VisualBlocks;

var mode = ThemeMode.Light;
var theme = VisualTheme.Graphite();
var model = PictorialRecipe.Create("options", mode);
var context = new VisualRenderContext(
    new VisualLayoutOptions(new VisualSize(800, 177)), theme, mode,
    new VisualFrame("Completed assessments by team", "", showLegend: null, showSurface: false));
var prepared = model.Prepare(context);
File.WriteAllText("visual.svg", prepared.ToSvg());
File.WriteAllBytes("visual.png", prepared.ToPng());
File.WriteAllText("visual.html", prepared.ToArtifact("example", VisualArtifactKind.Chart).ToHtmlPage());
#endregion

/// <summary>Assessment counts shown through repeated person symbols.</summary>
public static class PictorialRecipe {
    public static Chart Create(string variant = "wide", ThemeMode mode = ThemeMode.Light) {
        var chart = Chart.Create().AddPictorial("Assessments", new[] {
            new ChartPictorialItem("Team A", 74), new ChartPictorialItem("Team B", 46), new ChartPictorialItem("Team C", 61)
        }, ChartPictorialShape.Person).Pictorial(p => p.Max(100).ValuePerSymbol(10).Columns(10));
        if (variant == "options") chart.Pictorial(p => p.SvgPath("M2 2H22V22H2Z M8 8V16H16V8Z", new ChartRect(0, 0, 24, 24)));
        return chart;
    }
}
