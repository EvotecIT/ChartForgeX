#region Setup
using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Linq;
using ChartForgeX.Core;
using ChartForgeX.Primitives;
using ChartForgeX.Rendering;
using ChartForgeX.Themes;
using ChartForgeX.Topology;
using ChartForgeX.Typography;
using ChartForgeX.VisualArtifacts;
using ChartForgeX.VisualBlocks;
using static RegionMapRecipe;

var mode = VisualThemeMode.Dark;
var theme = VisualTheme.Graphite();
#endregion

var chart = Temperature(mode);

#region Output
var context = new VisualRenderContext(
    new VisualLayoutOptions(new VisualSize(360, 360)), theme, mode,
    new VisualFrame("Temperature by state", "Temperature captions use their own format and culture", showLegend: null, showSurface: false));
var prepared = chart.Prepare(context);
File.WriteAllText("visual.svg", prepared.ToSvg());
File.WriteAllBytes("visual.png", prepared.ToPng());
File.WriteAllText("visual.html", prepared.ToArtifact("example", VisualArtifactKind.Chart).ToHtmlPage());
#endregion

static class RegionMapRecipe {
    internal static Chart Temperature(VisualThemeMode mode) {
        var colors = VisualTheme.Graphite().Resolve(mode);
        var values = new[] {
            new ChartRegionMapItem("CA", -25), new ChartRegionMapItem("TX", -3), new ChartRegionMapItem("NY", 0),
            new ChartRegionMapItem("FL", 2.5), new ChartRegionMapItem("WA", 5), new ChartRegionMapItem("IL", 22.5),
            new ChartRegionMapItem("CO", 55)
        };
        return Chart.Create().ConfigureLabels(labels => labels.NoData = "Not sampled")
            .AddRegionHeatmap("Daily mean", ChartMapCatalog.Get("us-states"), values,
                ChartColorScale.Diverging(ChartColor.FromHex("#2166AC"), ChartColor.FromHex("#D1D5DB"), ChartColor.FromHex("#B2182B"), 0)
                    .WithValueRange(-20, 40).WithLabels("Cold", "Freezing", "Warm").WithNoDataColor(colors.Border),
                new ChartSeriesId("state-temperatures"))
            .WithMapLabels(false).WithMapScaleLegendPosition(ChartMapScaleLegendPosition.Right)
            .ConfigureMapColorDimension(options => {
                options.Name = "Temperature";
                options.ValueFormat = ChartValueFormat.Number("#,0.### '°C'", System.Globalization.CultureInfo.InvariantCulture);
            });
    }
}
