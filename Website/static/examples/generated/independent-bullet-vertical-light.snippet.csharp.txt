#region Setup
using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Linq;
using ChartForgeX.Core;
using ChartForgeX.Primitives;
using ChartForgeX.Rendering;
using ChartForgeX.Themes;
using ChartForgeX.Topology;
using ChartForgeX.Typography;
using ChartForgeX.VisualArtifacts;
using ChartForgeX.VisualBlocks;

var mode = VisualThemeMode.Light;
var theme = VisualTheme.Graphite();
#endregion

var load = ChartSeries.CreateBullet(new ChartSeriesId("load"), "Capacity", new ChartBulletValue(.72, .8, 0, 1, new[] { .5, .8 }))
    .WithObservationId(0, new ChartObservationId("current"))
    .ConfigureBulletAxis(axis => { axis.WithValueFormat(ChartValueFormat.Number("0%")); axis.Labels.Add(new ChartAxisLabel(0, "0%")); axis.Labels.Add(new ChartAxisLabel(.5, "50%")); axis.Labels.Add(new ChartAxisLabel(1, "100%")); });
var queue = ChartSeries.CreateBullet(new ChartSeriesId("queue"), "Queue", new ChartBulletValue(1250, 900, 100, 1100, new[] { 500d, 900 }))
    .WithObservationId(0, new ChartObservationId("current"))
    .ConfigureBulletAxis(axis => { axis.Reversed = true; axis.TickCount = 3; axis.WithValueFormat(ChartValueFormat.Number("0' jobs'")); });
var chart = Chart.Create().WithSize(760, 480).WithTitle("Independent operating measurements")
    .WithTheme(VisualTheme.Graphite(), mode).WithLegend(false).AddSeries(load).AddSeries(queue)
    .ConfigureBullet(options => { options.IndependentAxes = true; options.Orientation = ChartOrientation.Vertical; });

#region Output
var context = new VisualRenderContext(
    new VisualLayoutOptions(new VisualSize(760, 480)), theme, mode,
    new VisualFrame("Independent operating measurements", "Separate domains and source units share horizontal or vertical tracks", showLegend: null, showSurface: false));
var prepared = chart.Prepare(context);
File.WriteAllText("visual.svg", prepared.ToSvg());
File.WriteAllBytes("visual.png", prepared.ToPng());
File.WriteAllText("visual.html", prepared.ToArtifact("example", VisualArtifactKind.Chart).ToHtmlPage());
#endregion
