#region Setup
using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Linq;
using ChartForgeX.Core;
using ChartForgeX.Primitives;
using ChartForgeX.Rendering;
using ChartForgeX.Themes;
using ChartForgeX.Topology;
using ChartForgeX.Typography;
using ChartForgeX.VisualArtifacts;
using ChartForgeX.VisualBlocks;

var mode = VisualThemeMode.Dark;
var theme = VisualTheme.Graphite();
#endregion

var colors = VisualTheme.Graphite().Resolve(mode);
var codes = new[] { "CA", "TX", "NY", "FL", "WA", "IL", "CO" };
var values = new[] { -25d, -3, 0, 2.5, 5, 22.5, 55 };
var scale = ChartColorScale.Diverging(new[] {
    new ChartColorStop(-20, ChartColor.FromHex("#2166AC")), new ChartColorStop(-3, ChartColor.FromHex("#67A9CF"))
}, new ChartColorStop(0, ChartColor.FromHex("#D1D5DB")), new[] {
    new ChartColorStop(5, ChartColor.FromHex("#D9A441")), new ChartColorStop(40, ChartColor.FromHex("#B2182B"))
}).WithLabels("Cold", "Freezing", "Warm").WithNoDataColor(colors.Border);
var chart = Chart.Create().ConfigureLabels(labels => labels.NoData = "Not sampled")
    .AddTileMap("Daily mean", ChartTileMapCatalog.Get("us-states"),
        codes.Select((code, index) => new ChartRegionMapItem(code, values[index])), id: new ChartSeriesId("state-temperatures"))
    .WithMapLabels().WithMapColorScale(scale)
    .ConfigureMapColorDimension(options => {
        options.Name = "Temperature";
        options.ValueFormat = ChartValueFormat.Number("#,0.### '°C'", System.Globalization.CultureInfo.InvariantCulture);
        options.LegendStyle = ChartColorLegendStyle.Gradient;
    });

#region Output
var context = new VisualRenderContext(
    new VisualLayoutOptions(new VisualSize(800, 440)), theme, mode,
    new VisualFrame("Temperature by state", "A continuous key retains every authored color anchor", showLegend: null, showSurface: false));
var prepared = chart.Prepare(context);
File.WriteAllText("visual.svg", prepared.ToSvg());
File.WriteAllBytes("visual.png", prepared.ToPng());
File.WriteAllText("visual.html", prepared.ToArtifact("example", VisualArtifactKind.Chart).ToHtmlPage());
#endregion
