/// <summary>Regional requests and follow-ups encoded as numeric radial columns.</summary>
public static class RadialColumnRecipe {
    /// <summary>Creates radial columns with optional normalized stacking or explicit axis-caption density.</summary>
    public static Chart Create(string variant = "wide", ThemeMode mode = ThemeMode.Light) {
        if (variant is not ("wide" or "compact" or "sparse" or "options" or "compact-options" or "axis-labels" or "compact-axis-labels" or "all-labels" or "compact-all-labels"))
            throw new ArgumentOutOfRangeException(nameof(variant));
        if (variant is "all-labels" or "compact-all-labels") return AllLabels();
        var requests = new ChartSeries(new ChartSeriesId("requests"), "Requests", ChartSeriesKind.RadialColumn, Points(1200, 950, 680, 1050));
        var followUps = new ChartSeries(new ChartSeriesId("follow-ups"), "Follow-ups", ChartSeriesKind.RadialColumn, Points(320, 410, 260, 360));
        var chart = Chart.Create().XLabels("North", "South", "East", "West")
            .RadialGeometry(new ChartRadialGeometryOptions(-90, 180, .22, .22, .15))
            .AddSeries(requests).AddSeries(followUps).DataLabels().DataLabels(d => d.Placement(ChartDataLabelPlacement.Inside));
        if (variant is "options" or "compact-options") {
            requests.StackGroup("work").NormalizedTo(100);
            followUps.StackGroup("work").NormalizedTo(100).PointFillPattern(1, ChartFillPattern.DiagonalForward);
            chart.RadialGeometry(new ChartRadialGeometryOptions(-90, 180, .22, .22, .15, cornerRadius: 6))
                .YAxis(y => y.Bounds(0, 100)).XAxis(x => x.Title("Region")).YAxis(y => y.Title("Share of regional workload (%)"))
                .XAxis(axis => axis.LabelAngle = -30).YAxis(axis => axis.LabelAngle = 30);
        } else {
            chart.YAxis(y => y.Bounds(0, 1500));
        }
        if (variant is "axis-labels" or "compact-axis-labels") {
            chart.XAxis(x => x.Title("Region")).YAxis(y => y.Title("Requests"))
                .XAxis(axis => { axis.LabelOrientation = ChartAxisLabelOrientation.Parallel; axis.LabelAngle = 10; axis.LabelDensity = ChartLabelDensity.Relaxed; })
                .YAxis(axis => { axis.LabelOrientation = ChartAxisLabelOrientation.Perpendicular; axis.LabelDensity = ChartLabelDensity.Dense; });
        }
        return chart;
    }

    private static Chart AllLabels() {
        var requests = new ChartSeries(new ChartSeriesId("requests"), "Requests", ChartSeriesKind.RadialColumn,
            Enumerable.Range(1, 20).Select(index => new ChartPoint(index, 500 + index * 20)));
        var chart = Chart.Create().XLabels(Enumerable.Range(1, 20).Select(index => "Region " + index + " request volume").ToArray())
            .RadialGeometry(new ChartRadialGeometryOptions(-75, 195, .2, .25, .1))
            .YAxis(y => y.Bounds(0, 1500)).DataLabels(false).XAxis(x => x.Title("All region captions; overlaps are intentional"))
            .XAxis(axis => axis.LabelDensity = ChartLabelDensity.All).AddSeries(requests);
        chart.Options.YAxis.Visible = false;
        return chart;
    }

    private static ChartPoint[] Points(params double[] values) => values.Select((value, index) => new ChartPoint(index + 1, value)).ToArray();
}
