#region Setup
using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Linq;
using ChartForgeX.Core;
using ChartForgeX.Primitives;
using ChartForgeX.Rendering;
using ChartForgeX.Themes;
using ChartForgeX.Topology;
using ChartForgeX.Typography;
using ChartForgeX.VisualArtifacts;
using ChartForgeX.VisualBlocks;

var mode = VisualThemeMode.Light;
var theme = VisualTheme.Graphite();
#endregion

Chart chart;
chart = Chart.Create().WithTitle("Signed delivery intervals")
    .WithXLabels(Enumerable.Range(1, 24).Select(index => "R" + index.ToString("00", CultureInfo.InvariantCulture)).ToArray())
    .WithBarStyle(ChartBarStyle.Flat).WithDataLabels()
    .ConfigureRangeBar(options => options.Orientation = ChartOrientation.Vertical)
    .AddSeries(ChartSeries.CreateRangeBar(new ChartSeriesId("range"), "Delivery range", Enumerable.Range(1, 24).Select(index => {
        var start = index % 6 == 0 ? 0 : Math.Sin(index * .9) * 20;
        return new ChartInterval(index, start, index % 6 == 0 ? 0 : start + (index % 3 == 0 ? -10 : 10));
    })));

#region Output
var context = new VisualRenderContext(
    new VisualLayoutOptions(new VisualSize(390, 320)), theme, mode,
    new VisualFrame("Minimum and maximum delivery times", "", showLegend: null, showSurface: false));
var prepared = chart.Prepare(context);
File.WriteAllText("visual.svg", prepared.ToSvg());
File.WriteAllBytes("visual.png", prepared.ToPng());
File.WriteAllText("visual.html", prepared.ToArtifact("example", VisualArtifactKind.Chart).ToHtmlPage());
#endregion
